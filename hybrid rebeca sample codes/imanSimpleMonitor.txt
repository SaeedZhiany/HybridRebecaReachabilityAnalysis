physicalclass Monitor(3) {
    knownrebecs {}
    statevars {float timer; float r;}
    Monitor() {
        setMode(Runningg);
        r = 0.0f;
    }

    msgsrv start() {
        timer = 0.0f;
        r = 1.0f;
    }

    msgsrv stop() {
        r = 0.0f;
    }

    mode Runningg{
        inv(true || true){
            timer' = r;
        }

        guard(false && false){
        }
    }
}

physicalclass WheelWithSensor (3){
    knownrebecs {WheelCtlr ctlr; Monitor monitor;}
    statevars {float trq; float spd; float t;}
    WheelWithSensor(float spd_){
        spd = spd_;
        setMode(Rolling);
    }

    msgsrv setTrq(float trq_){
        trq = trq_;
        monitor.stop();
    }

    mode Rolling{
        inv(t <= 0.05){
            t' = 1;
            spd' = -1.0f - trq;
        }

        guard(t == 0.05){
            t = 0.0f;
            ctlr.setWspd(spd);
            if(spd > 0) setMode(Rolling);
            else setMode(none);
        }
    }
}

reactiveclass WheelCtlr (3){
    knownrebecs {WheelWithSensor w; BrakeCtlr bctlr;}
    statevars {int id; float wspd; float slprt; float WRAD;}
    WheelCtlr(int id_){
        id = id_;
        WRAD = 0.3f;
    }

    msgsrv setWspd(float wspd_){
        wspd = wspd_;
        bctlr.setWspd(wspd) after(0.011, 0.011);
    }

    msgsrv applyTrq(float reqTrq, float vspd){
        if(vspd == 0) slprt = 0.0f;
        else slprt = (vspd - wspd*WRAD) /vspd;
        if(slprt > 0.2) w.setTrq(0.0f);
        else w.setTrq(reqTrq);
    }
}

physicalclass BrakeWithSensor (3){
    knownrebecs {BrakeCtlr bctlr; Monitor monitor;}
    statevars {float bprcnt; float t;
    float mxprcnt; float r;}
    BrakeWithSensor(float b_, float mx_){
        bprcnt = b_;
        mxprcnt = mx_;
        r = 1.0f;
        setMode(Braking);
    }

    mode Braking{
        inv(t <= 0.05){
            t' = 1;
            bprcnt' = r;
        }

        guard(t == 0.05){
            t = 0.0f;
            bctlr.setBprcnt(bprcnt);
            monitor.start();
            if(bprcnt >= mxprcnt) r = 0.0f;
        }
    }
}

reactiveclass BrakeCtlr (3){
    knownrebecs {WheelCtlr wctlrR;}
    statevars {float wspdR; float bprcnt; float gtrq; float espd;}

    msgsrv control(){
        espd=wspdR;
        gtrq =bprcnt;
        wctlrR.applyTrq(gtrq, espd) after(0.011, 0.011);
    }

    msgsrv setWspd(float wspd_){
        wspdR = wspd_;
    }

    msgsrv setBprcnt(float bprcnt_){
        bprcnt = bprcnt_;
    }
}

physicalclass Clock (3){
    knownrebecs {BrakeCtlr bctlr;}
    statevars {float t;}

    Clock(){
        setMode(Running);
    }

    mode Running{
        inv(t <=0.05){
            t' = 1;
        }
        guard(t == 0.05){
            t = 0.0f;
            bctlr.control();
        }
    }
}

main {
    WheelWithSensor RightWheel(RightWCtlr, monitorr):(1.0f);
    Monitor monitorr():();
    WheelCtlr RightWCtlr(RightWheel, bctlr):(0);
    BrakeCtlr bctlr(RightWCtlr):();
    BrakeWithSensor brake(bctlr, monitorr):(60.0f,65.0f);
    Clock clock(bctlr):();
}
