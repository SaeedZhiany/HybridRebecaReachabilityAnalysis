physicalclass WheelWithSensor (3){
    knownrebecs {WheelCtlr ctlr;}
    statevars {float trq; float spd; float t;}
    WheelWithSensor(float spd_){
        spd = spd_;
        setMode(Rolling);
    }

    msgsrv setTrq(float trq_){
        trq = trq_;
    }

    mode Rolling{
        inv(t <= 0.05){
            t' = 1;
            spd' = -1.0f - trq;
        }

        guard(t >= 0.05){
            t = 0;
            ctlr.setWspd(spd);
            if(spd > 0) setMode(Rolling);
            else setMode(none);
        }
    }
}

reactiveclass WheelCtlr (3){
    knownrebecs {WheelWithSensor w; BrakeCtlr bctlr;}
    statevars {int id; float wspd; float slprt; float WRAD;}
    WheelCtlr(int id_){
        id = id_;
        WRAD = 0.3f;
    }

    msgsrv setWspd(float wspd_){
        wspd = wspd_;
        bctlr.setWspd(id,wspd) after(0.011, 0.011);
    }

    msgsrv applyTrq(float reqTrq, float vspd){
        if(vspd == 0) slprt = 0;
        else slprt = (vspd - wspd*WRAD) /vspd;
        if(slprt > 0.2) w.setTrq(0);
        else w.setTrq(reqTrq);
    }
}

physicalclass BrakeWithSensor (3){
    knownrebecs {BrakeCtlr bctlr;}
    statevars {float bprcnt; float t;
    float mxprcnt; float r;}
    BrakeWithSensor(float b_, float mx_){
        bprcnt = b_;
        mxprcnt = mx_;
        r = 1;
        setMode(Braking);
    }

    mode Braking{
        inv(t <= 0.05){
            t' = 1;
            bprcnt' = r;
        }

        guard(t == 0.05){
            t = 0;
            bctlr.setBprcnt(bprcnt);
            if(bprcnt >= mxprcnt) r = 0;
            setMode(Braking);
        }
    }
}

reactiveclass BrakeCtlr (3){
    knownrebecs {WheelCtlr wctlrR; WheelCtlr wctlrL;}
    statevars {float wspdR; float wspdL; float bprcnt; float gtrq; float espd;}
    
    msgsrv control(){
        espd=(wspdR+wspdL)/2;
        gtrq =bprcnt;
        wctlrR.applyTrq(gtrq, espd) after(0.011, 0.011);
        wctlrL.applyTrq(gtrq, espd) after(0.011, 0.011);
    }
        
    msgsrv setWspd(int id_, float wspd_){
        if(id_ == 0) wspdR = wspd_;
        else wspdL = wspd_;
    } 

    msgsrv setBprcnt(float bprcnt_){
        bprcnt = bprcnt_;
    }
}

physicalclass Clock (3){
    knownrebecs {BrakeCtlr bctlr;}
    statevars {float t;}

    Clock(){
        setMode(Running);
    }

    mode Running{
        inv(t <=0.05){
            t' = 1;
        }
        guard(t == 0.05){
            t = 0;
            bctlr.control();
            setMode(Running);
        }
    }
}

main {
    WheelWithSensor RightWheel(RightWCtlr):(1.0f);
    WheelWithSensor LeftWheel(LeftWCtlr):(1.0f);
    WheelCtlr RightWCtlr(RightWheel, bctlr):(0);
    WheelCtlr LeftWCtlr(LeftWheel, bctlr):(1);
    BrakeCtlr bctlr(RightWCtlr, LeftWCtlr):();
    BrakeWithSensor brake(bctlr):(60.0f,65.0f);
    Clock clock(bctlr):();
}
